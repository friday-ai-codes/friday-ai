## Task 1: Actor 提取中间件

### Prompt

Create ASGI middleware `AuditActorMiddleware` in `server/audit/middleware.py` that extracts actor info from the request context and stores it in a contextvars ContextVar.

**ContextVar setup** in `server/audit/context.py`:

```python
import contextvars
from dataclasses import dataclass

@dataclass
class AuditActor:
    user: object | None = None       # User instance or None
    actor_type: str = "user"         # user / system
    ip_address: str | None = None
    user_agent: str = ""

_current_actor: contextvars.ContextVar[AuditActor] = contextvars.ContextVar(
    "audit_actor", default=AuditActor()
)

def get_current_actor() -> AuditActor:
    return _current_actor.get()

def set_current_actor(actor: AuditActor) -> contextvars.Token:
    return _current_actor.set(actor)

def reset_current_actor(token: contextvars.Token) -> None:
    _current_actor.reset(token)
```

**Middleware** (ASGI-style, wraps scope):

```python
class AuditActorMiddleware:
    def __init__(self, app):
        self.app = app

    async def __call__(self, scope, receive, send):
        if scope["type"] not in ("http", "websocket"):
            return await self.app(scope, receive, send)

        actor = AuditActor()
        # Extract from scope (Django sets scope["user"] after auth middleware)
        user = scope.get("user")
        if user and hasattr(user, "is_authenticated") and user.is_authenticated:
            actor.user = user
            actor.actor_type = "user"
        else:
            actor.actor_type = "system"

        # IP
        if scope.get("client"):
            actor.ip_address = scope["client"][0]

        # User-Agent from headers
        for key, val in scope.get("headers", []):
            if key == b"user-agent":
                actor.user_agent = val.decode("utf-8", errors="replace")
                break

        token = set_current_actor(actor)
        try:
            return await self.app(scope, receive, send)
        finally:
            reset_current_actor(token)
```


## Task 2: emit_audit_event() 双入口

### Prompt

Create emit functions in `server/audit/emit.py` — sync + async dual entry, best-effort.

```python
import structlog
from audit.context import get_current_actor
from audit.models import AuditEvent

logger = structlog.get_logger("audit")

def emit_audit_event(
    action: str,
    target_type: str,
    target_id: str = "",
    before: dict | None = None,
    after: dict | None = None,
    *,
    actor=None,  # override auto-extraction
    source: str = AuditEvent.Source.API,
) -> AuditEvent | None:
    """Synchronous emit. Best-effort — returns None on failure, never raises."""
    try:
        ctx = get_current_actor()
        event = AuditEvent(
            action=action,
            target_type=target_type,
            target_id=str(target_id),
            before=before or {},
            after=after or {},
            source=source,
            actor=actor or ctx.user,
            actor_display=(actor or ctx.user).get_username() if (actor or ctx.user) else "system",
            actor_type=ctx.actor_type if not actor else "user",
            ip_address=ctx.ip_address,
            user_agent=ctx.user_agent,
        )
        event.save()
        logger.info("audit_event_created", action=action, target_type=target_type, target_id=target_id)
        return event
    except Exception:
        logger.warning("audit_event_emit_failed", action=action, target_type=target_type, target_id=target_id, exc_info=True)
        return None

async def aemit_audit_event(
    action: str,
    target_type: str,
    target_id: str = "",
    before: dict | None = None,
    after: dict | None = None,
    **kwargs,
) -> AuditEvent | None:
    """Async wrapper — delegates to sync via sync_to_async."""
    from asgiref.sync import sync_to_async
    return await sync_to_async(emit_audit_event)(
        action, target_type, target_id, before, after, **kwargs
    )
```
